import React from 'react';
import { motion } from 'framer-motion';
import { MessageSquare, Send, FileText, AlertCircle } from 'lucide-react';
import { useQuery } from '@tanstack/react-query';
import { fetchUsage } from '@/api/billingService';
import { useLanguage } from '@/lib/LanguageContext';
 
interface QuotaBarProps {
    label: string;
    current: number;
    limit: number;
    icon: React.ElementType;
    color: string;
}
 
const QuotaBar = ({ label, current, limit, icon: Icon, color }: QuotaBarProps) => {
    const percentage = Math.min(Math.round((current / limit) * 100), 100);
    const isHigh = percentage > 80;
 
    return (
        <div className="space-y-2">
            <div className="flex justify-between items-center text-[10px] font-bold uppercase tracking-wider">
                <div className="flex items-center gap-1.5 text-gray-400">
                    <Icon className="w-3 h-3" />
                    <span>{label}</span>
                </div>
                <span className={isHigh ? 'text-red-500' : 'text-gray-500'}>
                    {current} / {limit}
                </span>
            </div>
            <div className="h-1.5 w-full bg-gray-100 dark:bg-gray-800 rounded-full overflow-hidden border border-gray-200/50 dark:border-gray-700/50">
                <motion.div
                    initial={{ width: 0 }}
                    animate={{ width: `${percentage}%` }}
                    transition={{ duration: 1, ease: "easeOut" }}
                    className={`h-full rounded-full ${color} ${isHigh ? 'animate-pulse' : ''}`}
                />
            </div>
        </div>
    );
};
 
export const UsageQuota = ({ compact = false }: { compact?: boolean }) => {
    const { t } = useLanguage();
    const { data: usageResponse, isLoading, error } = useQuery({
        queryKey: ['usage'],
        queryFn: fetchUsage,
        refetchInterval: 60000, // Refetch every minute
    });
 
    const usage = usageResponse?.data?.usage;
    const plan = usageResponse?.data?.plan;
 
    if (isLoading) {
        return (
            <div className="p-4 space-y-4 animate-pulse">
                <div className="h-4 bg-gray-200 dark:bg-gray-800 rounded w-1/2"></div>
                <div className="space-y-3">
                    <div className="h-2 bg-gray-200 dark:bg-gray-800 rounded"></div>
                    <div className="h-2 bg-gray-200 dark:bg-gray-800 rounded"></div>
                </div>
            </div>
        );
    }
 
    if (error || !usage || !plan) {
        return null;
    }
 
    // Default limits if not in plan (fallback)
    const features = (plan.features as any) || {};
    const limits = {
        sms: features.smsLimit || 50,
        ai: features.aiChatLimit || 20,
        report: features.reportLimit || 5
    };
 
    return (
        <div className={`p-4 ${compact ? '' : 'bg-gray-50/50 dark:bg-gray-800/30 rounded-2xl border border-gray-100 dark:border-gray-700'}`}>
            {!compact && (
                <div className="flex items-center justify-between mb-4">
                    <h4 className="text-xs font-bold text-gray-900 dark:text-white uppercase tracking-wider font-mono">
                        {t('billing_quota_usage')}
                    </h4>
                    <span className="px-2 py-0.5 rounded-full bg-primary-100 dark:bg-primary-900/40 text-[9px] font-bold text-primary-600 dark:text-primary-400">
                        {plan.name}
                    </span>
                </div>
            )}
 
            <div className="space-y-4">
                <QuotaBar
                    label={t('nav_sms')}
                    current={usage.smsCount}
                    limit={limits.sms}
                    icon={Send}
                    color="bg-primary-500"
                />
                <QuotaBar
                    label={t('chat_ai_advisor')}
                    current={usage.aiChatCount}
                    limit={limits.ai}
                    icon={MessageSquare}
                    color="bg-secondary-500"
                />
                <QuotaBar
                    label={t('reports_title')}
                    current={usage.reportCount}
                    limit={limits.report}
                    icon={FileText}
                    color="bg-purple-500"
                />
            </div>
 
            {!compact && usage.smsCount / limits.sms > 0.9 && (
                <div className="mt-4 p-2 rounded-lg bg-red-50 dark:bg-red-900/20 border border-red-100 dark:border-red-800 flex items-start gap-2">
                    <AlertCircle className="w-3.5 h-3.5 text-red-500 mt-0.5" />
                    <p className="text-[10px] text-red-600 dark:text-red-400 leading-tight">
                        {t('billing_limit_warning')}
                    </p>
                </div>
            )}
        </div>
    );
};
